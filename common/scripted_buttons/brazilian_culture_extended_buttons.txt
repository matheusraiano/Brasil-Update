# Brazil Update+ - Integration Buttons for New Cultures
# Adds mineiro, carioca, and sertanejo to the Brazilian nation building system

je_brazilian_nation_building_integrate_mineiro = {
	name = {
		first_valid = {
			triggered_desc = {
				desc = "je_brazilian_nation_building_integrate_mineiro"
				trigger = {
					NOT = { has_variable = mineiros_integrated }
				}
			}
			triggered_desc = {
				desc = "je_brazilian_nation_building_integrate_mineiro_past_tense"
				trigger = {
					has_variable = mineiros_integrated
				}
			}
		}
	}
	desc = "je_brazilian_nation_building_integrate_mineiro_desc"

	visible = { always = yes }

	possible = {
		cu:brazilian.culture_current_fervor > cu:mineiro.culture_current_fervor
		any_scope_state = {
			state_region = {
				is_homeland = cu:mineiro
			}
		}
		OR = {
			custom_tooltip = {
				text = je_brazilian_nation_building_brazilian_national_hero_owned
				has_variable = brazilian_mineiro_hero_var
			}
			custom_tooltip = {
				text = je_brazilian_nation_building_mineiro_gold_mines
				any_scope_state = {
					state_region = {
						is_homeland = cu:mineiro
					}
					any_scope_building = {
						is_building_type = building_gold_mine
						level >= 10
					}
				}
			}
		}
		hidden_trigger = {
			NOT = {
				has_variable = mineiros_integrated
			}
		}
	}

	effect = {
		trigger_event = { id = brazil_culture_extended.1 }
		hidden_effect = {
			if = {
				limit = {
					has_variable = brazilian_mineiro_hero_var
				}
				remove_variable = brazilian_mineiro_hero_var
			}
			set_variable = {
				name = mineiros_integrated
			}
		}
		custom_tooltip = {
			text = je_brazilian_nation_building_integrate_mineiro_removal_tt
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:mineiro
					}
				}
				state_region = {
					remove_homeland = cu:mineiro
				}
			}
		}
	}

	ai_chance = {
		value = 100 # Always press this button
	}
}
