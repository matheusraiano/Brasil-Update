namespace = culture_brazil

# A New National Hero - Extended Version with New Cultures
culture_brazil.1 = {
	type = country_event
	placement = scope:state

	title = culture_brazil.1.t
	desc = culture_brazil.1.d
	flavor = culture_brazil.1.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:brazilian_national_hero

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	dlc = dlc008

	cooldown = { days = normal_modifier_time }

	trigger = {
		# triggered by on_action on_battle_won pulse
		has_journal_entry = je_brazilian_nation_building
		OR = {
			AND = {
				NOT = { has_variable = brazilian_sulista_hero_var }
				NOT = { has_variable = sulistas_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_nordestino_hero_var }
				NOT = { has_variable = nordestinos_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_amazonic_hero_var }
				NOT = { has_variable = amazonics_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_paulista_hero_var }
				NOT = { has_variable = paulistas_integrated }
			}
			# Brazil Update+ New Cultures
			AND = {
				NOT = { has_variable = brazilian_mineiro_hero_var }
				NOT = { has_variable = mineiros_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_carioca_hero_var }
				NOT = { has_variable = cariocas_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_sertanejo_hero_var }
				NOT = { has_variable = sertanejos_integrated }
			}
		}
		# there exists a commander on the root side to be the national hero
		OR = {
			scope:attacker = {
				owner = ROOT
			}
			scope:defender ?= {
				owner = ROOT
			}
		}
		scope:enemy_country = {
			NOR = {
				country_has_primary_culture = cu:brazilian
				country_has_primary_culture = cu:sulista
				country_has_primary_culture = cu:nordestino
				country_has_primary_culture = cu:amazonic
				country_has_primary_culture = cu:paulista
				country_has_primary_culture = cu:mineiro
				country_has_primary_culture = cu:carioca
				country_has_primary_culture = cu:sertanejo
			}
		}
	}

	immediate = {
		if = {
			limit = {
				scope:attacker = {
					owner = ROOT
				}
			}
			scope:attacker = {
				save_scope_as = brazilian_national_hero
			}
		}
		else = {
			scope:defender = {
				save_scope_as = brazilian_national_hero
			}
		}
		random_list = {
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_sulista_hero_var }
				 		NOT = { has_variable = sulistas_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:sulista
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_paulista_hero_var }
				 		NOT = { has_variable = paulistas_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:paulista
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_nordestino_hero_var }
				 		NOT = { has_variable = nordestinos_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:nordestino
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_amazonic_hero_var }
				 		NOT = { has_variable = amazonics_integrated }
				 	}
		 		}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:amazonic
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			# Brazil Update+ New Culture Heroes
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_mineiro_hero_var }
				 		NOT = { has_variable = mineiros_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:mineiro
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_carioca_hero_var }
				 		NOT = { has_variable = cariocas_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:carioca
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_sertanejo_hero_var }
				 		NOT = { has_variable = sertanejos_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:sertanejo
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
		}

		random_scope_character = {
			limit = { has_variable = brazilian_hero_scope }
			save_scope_as = brazilian_national_hero
			culture = {
				save_scope_as = brazilian_national_hero_culture
			}
		}
	}

	option = {
		name = culture_brazil.1.a
		default_option = yes

		custom_tooltip = culture_brazil.1.hero_clarification

		if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:sulista = this
					}
				}
			}
			set_variable = brazilian_sulista_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:nordestino = this
					}
				}
			}
			set_variable = brazilian_nordestino_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:amazonic = this
					}
				}
			}
			set_variable = brazilian_amazonic_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:paulista = this
					}
				}
			}
			set_variable = brazilian_paulista_hero_var
		}
		# Brazil Update+ Hero Variables
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:mineiro = this
					}
				}
			}
			set_variable = brazilian_mineiro_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:carioca = this
					}
				}
			}
			set_variable = brazilian_carioca_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:sertanejo = this
					}
				}
			}
			set_variable = brazilian_sertanejo_hero_var
		}

		scope:brazilian_national_hero = {
			if = {
				limit = {
					NOR = {
						has_trait = traditionalist_commander
						has_trait = popular_commander
						has_trait = celebrity_commander
					}
				}
				add_trait = traditionalist_commander
			}
			else_if = {
				limit = {
					has_trait = traditionalist_commander
				}
				remove_trait = traditionalist_commander
				add_trait = popular_commander
			}
			else_if = {
				limit = {
					has_trait = popular_commander
				}
				remove_trait = popular_commander
				add_trait = celebrity_commander
			}
		}
	}


	option = {
		name = culture_brazil.1.b

		scope:brazilian_national_hero = {
			retire_character = yes
		}
	}
}
